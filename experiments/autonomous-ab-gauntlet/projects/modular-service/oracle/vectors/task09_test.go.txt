package billing

import (
	"context"
	contract "example.com/acme/modular-service/internal/contracts/billing"
	"testing"
)

func TestGauntletRefundVector(t *testing.T) {
	s := New()
	if _, err := s.Issue(context.Background(), contract.CreateInvoiceRequest{OrderID: "O", AmountCents: 100}); err != nil {
		t.Fatal(err)
	}
	if err := s.Refund(context.Background(), "R0", "O", 0); err == nil {
		t.Fatal("zero refund accepted")
	}
	if err := s.Refund(context.Background(), "Rneg", "O", -1); err == nil {
		t.Fatal("negative refund accepted")
	}
	if err := s.Refund(context.Background(), "R1", "O", 40); err != nil {
		t.Fatal(err)
	}
	if err := s.Refund(context.Background(), "R1", "O", 40); err != nil {
		t.Fatalf("retry: %v", err)
	}
	if err := s.Refund(context.Background(), "R2", "O", 60); err != nil {
		t.Fatalf("remaining balance after idempotent retry: %v", err)
	}
	if err := s.Refund(context.Background(), "R3", "O", 1); err == nil {
		t.Fatal("refund exceeded remaining balance")
	}
	invoice, err := s.Issue(context.Background(), contract.CreateInvoiceRequest{OrderID: "O", AmountCents: 100})
	if err != nil || invoice.AmountCents != 100 {
		t.Fatalf("issued amount mutated: %+v %v", invoice, err)
	}
}
